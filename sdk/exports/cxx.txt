# The C++ runtime for ELF apps. Format: see viper.txt. Apps are built like the
# firmware: without exceptions or RTTI (docs/plans/elf_app_loader.md, Phase 4).
#
# These are mangled names, so they're tied to the toolchain the firmware and
# apps are built with (GCC 14, the C++11 string ABI, 32-bit size_t: "j" is
# unsigned int). A toolchain update can change them; check_app_elf.py catches
# that when an app is built.

# operator new and delete, tracked like malloc and free (esp_elf_alloc.h):
# new(size_t), new[](size_t), delete(void *), delete(void *, size_t), and the
# same for delete[]
_Znwj = esp_elf_app_new
_Znaj = esp_elf_app_new
_ZdlPv = esp_elf_app_free
_ZdlPvj = esp_elf_app_free
_ZdaPv = esp_elf_app_free
_ZdaPvj = esp_elf_app_free

# Global destructors, run when the app is unloaded (esp_elf_destruct())
__cxa_atexit = esp_elf_cxa_atexit

# Atomics (std::atomic, C11 atomics). Apps are compiled without hardware
# atomics, which don't work on PSRAM (elf_loader.cmake), so these are calls
# to ESP-IDF's versions. Aligned 4-byte loads and stores need no call.
__atomic_compare_exchange_1
__atomic_compare_exchange_4
__atomic_exchange_1
__atomic_exchange_4
__atomic_fetch_add_4
__atomic_fetch_and_4
__atomic_fetch_or_4
__atomic_fetch_sub_4

# Function-local statics: the firmware's thread-safe guards (ESP-IDF's
# cxx_guards.cpp). The guard variables are in the app.
__cxa_guard_acquire
__cxa_guard_release
__cxa_guard_abort

# The parts of libstdc++ that aren't header templates, so apps can't compile
# them in, and the prebuilt library can't be linked into an app (it isn't

# firmware (ESP-IDF's cxx_exception_stubs.cpp).

_ZSt20__throw_out_of_rangePKc

_ZSt28__throw_bad_array_new_lengthv
# Calling an empty std::function
_ZSt25__throw_bad_function_callv
# std::map and std::set
_ZSt18_Rb_tree_decrementPSt18_Rb_tree_node_base
