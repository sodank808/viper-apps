# libc for ELF apps, beyond the table in src/esp_elf_symbol.c. Format: see
# viper.txt.
#
# fopen and fclose (and opendir and closedir) go through
# components/system/src/elf_files.cpp, which closes the files and directories
# an app leaves open when it closes, as the loader frees the memory it doesn't.

# stdio.h: reading files
fopen = viper_elf_fopen
fclose = viper_elf_fclose
fread
fgets
fseek
ftell

# Writing files, for apps streaming into their data folder (viper_data_path()).
# fwrite, fputs and fprintf are in the libc table already.
fflush
remove
rename
# Formatting into a buffer, beside the table's snprintf (Doom's menus)
sprintf

# sys/stat.h and dirent.h. opendir and closedir go through elf_files.cpp too,
# so a directory left open is closed with the app.
mkdir
stat
# unistd.h: whether a file is there (Doom looks for its WADs and demos)
access
opendir = viper_elf_opendir
readdir
closedir = viper_elf_closedir

# pthread.h: what the libc table (pthread_create, _join, ...) lacks, for an
# app with a thread of its own to hand work to and from (Wolfenstein 3D's
# engine runs on one). Not tracked: an app destroys its mutexes and
# condition variables before it closes, or they leak (ESP-IDF allocates one
# for each on first use, statically initialized ones included).
pthread_mutex_init
pthread_mutex_destroy
pthread_mutex_lock
pthread_mutex_unlock
pthread_cond_init
pthread_cond_destroy
pthread_cond_signal
pthread_cond_broadcast
pthread_cond_wait
pthread_cond_timedwait

# string.h: what the libc table lacks. GCC also turns code into some of these
# on its own (a copy of a string literal into strcpy).
memchr
memcmp
memmove
strcasecmp
strcat
strcpy
strncasecmp
strncmp
strncpy
strstr
strtok

# stdlib.h: what the libc table lacks
atoi

# math.h: float math. Exporting one the firmware doesn't use yet adds its code

# libgcc, which apps link (elf_loader.cmake).

atanf
cbrtf
ceilf
cosf
exp2f
expf
fabsf
floorf
fmodf
hypotf
logf
lrintf
lround
lroundf
powf
roundf
sinf
sqrtf
tanf

# time.h: the game's clock, the calendar (full moon, Friday the 13th)
